# ── Transcription ────────────────────────────────────────────────────────────
faster-whisper>=1.0.0

# ── Forced alignment ─────────────────────────────────────────────────────────
# whisperx bundles wav2vec2 alignment and MMS models.
# Install AFTER torch, torchaudio and torchvision to avoid dependency resolution issues.
#
# Known harmless pip metadata conflicts (do not attempt to resolve):
#   whisperx 3.8.5 requires torchaudio~=2.8.0  — works fine with 2.9–2.11+
#   whisperx 3.8.5 requires torchvision~=0.23.0 — works fine with 0.24–0.26+
#   These are conservative pins in whisperx's metadata that lag PyTorch releases.
#   All pipeline functionality (whisperx.align, load_align_model) is unaffected.
whisperx>=3.0.0

# ── Voice Activity Detection ──────────────────────────────────────────────────
silero-vad>=5.0.0

# ── Audio I/O and signal processing ──────────────────────────────────────────

# ── Text normalisation ────────────────────────────────────────────────────────
num2words>=0.5.13

# ── Transcript reconciliation ─────────────────────────────────────────────────
diff-match-patch>=20230430

# ── Numeric / array utilities (transitive deps — pinned for stability) ────────
# Numba (used by librosa) requires NumPy <=2.2 — do not raise this ceiling
# until numba releases support for NumPy 2.3+
numpy>=1.24,<2.3
